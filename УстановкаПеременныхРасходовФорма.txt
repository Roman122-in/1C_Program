&НаКлиенте
Процедура Автозаполнение(Команда)
	
	Если Не ЗначениеЗаполнено(Объект.Дата) Тогда
		Сообщить("Сначала укажите дату документа!");
		Возврат;
	КонецЕсли;
	
	АвтозаполнениеНаСервере();
	
	Модифицированность = Истина;
	Сообщить("Табличная часть успешно заполнена!");
	
КонецПроцедуры

&НаСервере
Процедура АвтозаполнениеНаСервере()
	
	
	ДатаДок = Объект.Дата;
	НачалоПрошлогоМесяца = НачалоМесяца(ДобавитьМесяц(ДатаДок, -1));
	КонецПрошлогоМесяца = КонецМесяца(ДобавитьМесяц(ДатаДок, -1));
	
	ЗапросФакта = Новый Запрос;
	ЗапросФакта.Текст = 
		"ВЫБРАТЬ
		|   СУММА(ДеталиСмены.КоличествоСырьяКг) КАК ВсегоСырьяКг,
		|   СУММА(ДеталиСмены.КоличествоВедер) КАК ВсегоВедер
		|ИЗ
		|   Документ.СменныйРапорт.ДеталиСмены КАК ДеталиСмены
		|ГДЕ
		|   ДеталиСмены.Ссылка.Дата МЕЖДУ &Начало И &Конец";
		
	ЗапросФакта.УстановитьПараметр("Начало", НачалоПрошлогоМесяца);
	ЗапросФакта.УстановитьПараметр("Конец", КонецПрошлогоМесяца);
	
	Результат = ЗапросФакта.Выполнить().Выбрать();
	ВсегоСырьяКг = 0;
	ВсегоВедер = 0;
	
	Если Результат.Следующий() Тогда
		Если ЗначениеЗаполнено(Результат.ВсегоСырьяКг) Тогда ВсегоСырьяКг = Результат.ВсегоСырьяКг; КонецЕсли;;
		Если ЗначениеЗаполнено(Результат.ВсегоВедер) Тогда ВсегоВедер = Результат.ВсегоВедер; КонецЕсли;;
	КонецЕсли;
	
	СуммаСырье         = ВсегоСырьяКг * 33.0; 
	СуммаЭлектричество = ВсегоВедер * 1.51;   
	СуммаУпаковка      = ВсегоВедер * 0.65;   
	СуммаБрак          = 0;                   
	

	Для Каждого СтрокаТЧ Из Объект.УстановкаПеременныхРасходов Цикл
		
		Если Не ЗначениеЗаполнено(СтрокаТЧ.СтатьяРасходов) Тогда
			Продолжить;
		КонецЕсли;
		
		ИмяСтатьи = СтрокаТЧ.СтатьяРасходов.Наименование;
		РассчитаннаяСумма = 0;
		
		Если ИмяСтатьи = "Сырье" Или ИмяСтатьи = "Сырье (Вторичный п-н)" Тогда
			РассчитаннаяСумма = СуммаСырье;
		ИначеЕсли ИмяСтатьи = "Электричество" Или ИмяСтатьи = "Эле- во оборудования" Тогда
			РассчитаннаяСумма = СуммаЭлектричество;
		ИначеЕсли ИмяСтатьи = "Упаковка" Или ИмяСтатьи = "Упаковка и расходники" Тогда
			РассчитаннаяСумма = СуммаУпаковка;
		ИначеЕсли ИмяСтатьи = "Брак" Или ИмяСтатьи = "Переработка брака" Тогда
			РассчитаннаяСумма = СуммаБрак;
		КонецЕсли;
		
		
		СтрокаТЧ.Сумма = РассчитаннаяСумма;
		
	КонецЦикла;
	
КонецПроцедуры
