	Если Не ЗначениеЗаполнено(Объект.Аппарат) Или Не ЗначениеЗаполнено(Объект.Дата) Тогда

	ВыбранныйАппарат = Объект.Аппарат;

		Объект.Количество = СуммаЦиклов;
		
	ИначеЕсли НаименованиеАппарата = "Дробилка" Или НаименованиеАппарата = "Сушилка" Тогда
		
		
		Запрос = Новый Запрос;
		Запрос.Текст = 
			"ВЫБРАТЬ
			|   СУММА(ДеталиСмены.КоличествоСырьяКг) КАК ВсегоСырья

			|   Документ.СменныйРапорт.ДеталиСмены КАК ДеталиСмены
			|ГДЕ
			|   ДеталиСмены.Ссылка.Дата МЕЖДУ &НачалоДня И &КонецДня
			|   И ДеталиСмены.Ссылка.Аппарат = &Аппарат";
			
		Запрос.УстановитьПараметр("НачалоДня", НачалоДня(ЦелеваяДата));
		Запрос.УстановитьПараметр("КонецДня", КонецДня(ЦелеваяДата));
		Запрос.УстановитьПараметр("Аппарат", ВыбранныйАппарат);
		
		Результат = Запрос.Выполнить().Выбрать();
		Если Результат.Следующий() И ЗначениеЗаполнено(Результат.ВсегоСырья) Тогда
			Объект.Количество = Результат.ВсегоСырья;
