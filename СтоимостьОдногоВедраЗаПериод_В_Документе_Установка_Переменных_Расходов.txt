Процедура РасчетЕдиницы(Команда)
	
	Если Не ЗначениеЗаполнено(Объект.Дата) Тогда
		Сообщить("Укажите дату документа!");
		Возврат;
	КонецЕсли;
	
		ОтчетДанные = РассчитатьСебестоимостьНаСервере(Объект.Дата);
	
	Если ОтчетДанные.Ведра = 0 И ОтчетДанные.ОбщаяСумма = 0 Тогда
		ПоказатьПредупреждение(, "За прошлый месяц не найдено данных для расчета.");
		Возврат;
	КонецЕсли;
	
		ТекстСообщения = 
		"ИТОГИ РАСЧЕТА И ЭКОНОМИКА ЗА ПРОШЛЫЙ МЕСЯЦ:" + Символы.ПС +
		" " + Символы.ПС +
		"Всего ведер произведено: " + Формат(ОтчетДанные.Ведра, "ЧЦ=10; ЧДЦ=0") + Символы.ПС +
		"Переменные расходы (вкл. списание сырья): " + Формат(ОтчетДанные.Переменные, "ЧДЦ=2") + " грн." + Символы.ПС +
		"Постоянные расходы (аренда, ЗП, ТО): " + Формат(ОтчетДанные.Постоянные, "ЧДЦ=2") + " грн." + Символы.ПС +
		"Требуется на закупку нового сырья: " + Формат(ОтчетДанные.СуммаЗакупкиСырья, "ЧДЦ=2") + " грн." + Символы.ПС +
		"ВСЕГО РАСХОДОВ С УЧЕТОМ ЗАКУПКИ: " + Формат(ОтчетДанные.ПолныеРасходыСЗакупкой, "ЧДЦ=2") + " грн." + Символы.ПС +
		 " " + Символы.ПС +
		"Себестоимость : " + Формат(ОтчетДанные.СебестоимостьФакт, "ЧДЦ=2") + " грн./шт." + Символы.ПС +
		"Мин. цена без сырья : " + Формат(ОтчетДанные.МинЦенаБезубыточность, "ЧДЦ=2") + " грн./шт." + Символы.ПС +
		"Цена с учетом закупки сырья на следующий месяц. Целевая цена: " + Формат(ОтчетДанные.ЦелеваяЦенаПрибыль, "ЧДЦ=2") + " грн./шт." + Символы.ПС +
		"Прибыль при целевой цене: " + Формат(ОтчетДанные.ЧистаяПрибыльВМесяц, "ЧДЦ=2") + " грн.";
		
	ПоказатьПредупреждение(, ТекстСообщения);
	
КонецПроцедуры

&НаСервере
Функция РассчитатьСебестоимостьНаСервере(ДатаДок)
	
	НачалоПериода = НачалоМесяца(ДобавитьМесяц(ДатаДок, -1));
	КонецПериода = КонецМесяца(ДобавитьМесяц(ДатаДок, -1));
	
	Результат = Новый Структура;
	Результат.Вставить("Ведра", 0);
	Результат.Вставить("Переменные", 0);
	Результат.Вставить("Постоянные", 0);
	Результат.Вставить("ОбщаяСумма", 0);
	Результат.Вставить("СуммаЗакупкиСырья", 0);
	Результат.Вставить("ПолныеРасходыСЗакупкой", 0);
	Результат.Вставить("СебестоимостьФакт", 0);
	Результат.Вставить("МинЦенаБезубыточность", 0);
	Результат.Вставить("ЦелеваяЦенаПрибыль", 0);
	Результат.Вставить("ЧистаяПрибыльВМесяц", 0);
	
		ЗапросВедер = Новый Запрос;
	ЗапросВедер.Текст = 
		"ВЫБРАТЬ 
		|   ЕСТЬNULL(СУММА(ДеталиСмены.КоличествоВедер), 0) КАК ВсегоВедер 
		|    ИЗ 

		|   Документ.СменныйРапорт.ДеталиСмены КАК ДеталиСмены 
		|   ГДЕ 

		|   ДеталиСмены.Ссылка.Дата МЕЖДУ &Начало И &Конец";
		
	ЗапросВедер.УстановитьПараметр("Начало", НачалоПериода);
	ЗапросВедер.УстановитьПараметр("Конец", КонецПериода);
	
	РезВедер = ЗапросВедер.Выполнить().Выбрать();
	Если РезВедер.Следующий() Тогда
		Результат.Ведра = РезВедер.ВсегоВедер;
	КонецЕсли;
	
	Если Результат.Ведра = 0 Тогда 
		Возврат Результат; 
	КонецЕсли;
	
		ВыборкаПеременных = Документы.УстановкаПеременныхРасходов.Выбрать();
	Пока ВыборкаПеременных.Следующий() Цикл
		Если ВыборкаПеременных.Проведен И ВыборкаПеременных.Дата >= НачалоПериода И ВыборкаПеременных.Дата <= КонецПериода Тогда
			Результат.Переменные = Результат.Переменные + ВыборкаПеременных.Число;
		КонецЕсли;
	КонецЦикла;
	
		ВыборкаПостоянных = Документы.УстановкаПостоянныхРасходов.Выбрать();
	Пока ВыборкаПостоянных.Следующий() Цикл
		Если ВыборкаПостоянных.Проведен И ВыборкаПостоянных.Дата >= НачалоПериода И ВыборкаПостоянных.Дата <= КонецПериода Тогда
			Результат.Постоянные = Результат.Постоянные + ВыборкаПостоянных.Число;
		КонецЕсли;
	КонецЦикла;
	
		ЦенаЗаКгСырья = 33.00;
	МассаОдногоВедраКг = 0.35;
	ТребуемыйОбъемСырьяКг = Результат.Ведра * МассаОдногоВедраКг;
	Результат.СуммаЗакупкиСырья = ТребуемыйОбъемСырьяКг * ЦенаЗаКгСырья;
	
	Результат.ОбщаяСумма = Результат.Переменные + Результат.Постоянные;
	Результат.СебестоимостьФакт = Результат.ОбщаяСумма / Результат.Ведра;
	
		ДругиеПеременные = Макс(0, Результат.Переменные - Результат.СуммаЗакупкиСырья); 
	Результат.ПолныеРасходыСЗакупкой = Результат.Постоянные + Результат.СуммаЗакупкиСырья + ДругиеПеременные;
	
		Результат.МинЦенаБезубыточность = Результат.ПолныеРасходыСЗакупкой / Результат.Ведра;
	
		Результат.ЦелеваяЦенаПрибыль = Результат.МинЦенаБезубыточность * 1.25;
	
		
	ОбщаяВыручкаПриЦелевойЦене = Результат.ЦелеваяЦенаПрибыль * Результат.Ведра;
	Результат.ЧистаяПрибыльВМесяц = ОбщаяВыручкаПриЦелевойЦене - Результат.ПолныеРасходыСЗакупкой;
	
	Возврат Результат;
	
КонецФункции
